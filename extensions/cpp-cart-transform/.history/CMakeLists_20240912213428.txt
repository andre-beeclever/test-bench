
cmake_minimum_required(VERSION 3.10)
project(function VERSION 1.0.0)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

file(GLOB_RECURSE  SOURCES
    src/*.h
    src/*.hpp
    src/*.cpp
)

SET(CMAKE_CXX_FLAGS  "--target=wasm32-wasi --sysroot ~/wasi-sdk-24.0-x86_64-macos/share/wasi-sysroot/  -Oz -fno-exceptions -flto")
SET(CMAKE_EXE_LINKER_FLAGS  "-Wl,--strip-all -Wl,--gc-sections")
include_directories(${CMAKE_PROJECT_NAME} PRIVATE 
    ${PROJECT_SOURCE_DIR}/include
)

add_executable(${CMAKE_PROJECT_NAME} ${SOURCES})
