
cmake_minimum_required(VERSION 3.11)
project(function VERSION 1.0.0)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

file(GLOB_RECURSE SOURCES
    src/*.h
    src/*.hpp
    src/*.cpp
)

include_directories(${CMAKE_PROJECT_NAME} PRIVATE 
    ${PROJECT_SOURCE_DIR}/include
)
# enable FetchContent
include(FetchContent)

FetchContent_Declare(
  nlohmann/json
  GIT_REPOSITORY https://github.com/nlohmann/json.git
)

FetchContent_MakeAvailable(nlohmann/json)

SET(CMAKE_CXX_FLAGS  "-Oz -fno-exceptions -flto")
SET(CMAKE_EXE_LINKER_FLAGS  "-Wl,--strip-all -Wl,--gc-sections")
set(CMAKE_EXECUTABLE_SUFFIX ".wasm")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${PROJECT_SOURCE_DIR}/dist)
add_executable(${CMAKE_PROJECT_NAME} ${SOURCES})
